                        if (transposedIndex < 0) transposedIndex += 12;

                function parseLineWithChords(line) {
                    const items = [];
                    const regex = /\[(.*?)\]/g;
                    let lastIdx = 0;
                    let match;

                    while ((match = regex.exec(line)) !== null) {
                        const chord = match[1];
                        const chordIdx = match.index;

                        // Text before this chord
                        if (chordIdx > lastIdx) {
                            const precedingText = line.substring(lastIdx, chordIdx);
                            if (items.length > 0 && !items[items.length - 1].text) {
                                items[items.length - 1].text = precedingText;
                            } else {
                                items.push({ chord: '', text: precedingText });
                            }
                        }

                        items.push({ chord: chord, text: '' });
                        lastIdx = regex.lastIndex;
                    }

                    if (lastIdx < line.length) {
                        const remainingText = line.substring(lastIdx);
                        if (items.length > 0 && !items[items.length - 1].text) {
                            items[items.length - 1].text = remainingText;
                        } else {
                            items.push({ chord: '', text: remainingText });
                        }
                    }

                    return items;
                }

                function startMetronome() {
                    if (!audioCtx) {
                        audioCtx = new (window.AudioContext || window.webkitAudioContext)();
                    }
                    const bpm = activeSong.value?.bpm || 120;
                    const intervalMs = (60 / bpm) * 1000;

                    playTick();
                    metronomeTimer = setInterval(playTick, intervalMs);
                }

                function stopMetronome() {
                    if (metronomeTimer) clearInterval(metronomeTimer);
                }

                function playTick() {
                    if (!audioCtx) return;
                    const osc = audioCtx.createOscillator();
                    const gain = audioCtx.createGain();
                    osc.type = 'sine';
                    osc.frequency.setValueAtTime(800, audioCtx.currentTime);
                    gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
                    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.05);

                    osc.connect(gain);
                    gain.connect(audioCtx.destination);
                    osc.start();
                    osc.stop(audioCtx.currentTime + 0.05);
                }

                function showToast(msg) {
                    toastMessage.value = msg;
                    setTimeout(() => { toastMessage.value = ''; }, 2500);
                }
